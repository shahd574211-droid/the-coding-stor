# انسخ هذه الأسطر إلى ملف .env أو .env.local ثم غيّر القيم.

# ——— 1) مفتاح التشفير (لتسجيل الدخول + توكن التسجيل) ———
ENCRYPTION_KEY=your-secret-key-at-least-32-characters-long-change-me

# ——— 2) Supabase (مطلوب لإنشاء الحساب وتسجيل الدخول) ———
# احصل على القيم من: https://supabase.com/dashboard/project/_/settings/api
# استبدل your-project-ref بمعرّف مشروعك الحقيقي من لوحة Supabase.

NEXT_PUBLIC_SUPABASE_URL=https://your-project-ref.supabase.co
NEXT_PUBLIC_SUPABASE_ANON_KEY=your-anon-key-here
SUPABASE_SERVICE_ROLE_KEY=your-service-role-key-here

# مثال: إذا كان معرّف مشروعك abcdefgh فإن الرابط يكون:
# NEXT_PUBLIC_SUPABASE_URL=https://abcdefgh.supabase.co

# ——— 3) خصم عام على كل المنتجات (اختياري) ———
# نسبة مئوية 0–100. مثال: 10 = خصم 10٪ لجميع المنتجات. اتركه غير معرّف أو 0 لعدم وجود خصم.
# GLOBAL_DISCOUNT_PERCENT=10
